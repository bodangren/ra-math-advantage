const {chromium}=require('/Users/daniel.bodanske/Desktop/ra-math-advantage/node_modules/playwright');
const fs=require('fs');
const base='http://127.0.0.1:43132';
(async()=>{
 const browser=await chromium.launch({headless:true,channel:'chrome'});
 const student=await browser.newContext({storageState:'/tmp/precalc-q36-student-browser-state.json'});
 const sp=await student.newPage(); const studentErrors=[]; sp.on('console',m=>{if(m.type()==='error')studentErrors.push(m.text())});
 await sp.goto(base+'/student/lessons/1.1'); await sp.waitForLoadState('networkidle');
 const review=sp.getByRole('button',{name:'Mark worked example reviewed',exact:true});
 if(await review.count()!==1) throw new Error('Expected one strict Worked review control');
 await review.click(); await sp.waitForLoadState('networkidle'); await sp.waitForTimeout(700);
 await sp.screenshot({path:'/tmp/precalc-q36-student-after-worked.png',fullPage:true});
 const afterWorked=await sp.locator('body').innerText();
 const teacher=await browser.newContext({storageState:'/tmp/precalc-q36-teacher-browser-state.json'});
 const tp=await teacher.newPage(); const teacherErrors=[];tp.on('console',m=>{if(m.type()==='error')teacherErrors.push(m.text())});
 await tp.goto(base+'/teacher/dashboard');await tp.waitForLoadState('networkidle');
 await tp.screenshot({path:'/tmp/precalc-q36-teacher-candidate.png',fullPage:true});
 const candidateText=await tp.locator('body').innerText();
 fs.writeFileSync('/tmp/precalc-q36-browser-phase1.json',JSON.stringify({afterWorked,candidateText,studentErrors,teacherErrors},null,2));
 console.log('student buttons',await sp.getByRole('button').allTextContents());
 console.log('teacher buttons',await tp.getByRole('button').allTextContents());
 console.log('teacher selects',await tp.locator('select').count());
 console.log('candidate present',candidateText.includes('Q36 Local Student'));
 await student.close();await teacher.close();await browser.close();
})().catch(e=>{console.error(e);process.exit(1)});
