const {chromium}=require('/Users/daniel.bodanske/Desktop/ra-math-advantage/node_modules/playwright');const fs=require('fs');const base='http://127.0.0.1:43132';
(async()=>{const b=await chromium.launch({headless:true,channel:'chrome'});
const tc=await b.newContext({storageState:'/tmp/precalc-q36-teacher-browser-state.json'});const tp=await tc.newPage();await tp.goto(base+'/teacher/dashboard');await tp.waitForLoadState('networkidle');const confirm=tp.getByRole('button',{name:'Confirm classroom introduction',exact:true});if(await confirm.count()!=1)throw Error('expected confirm');await confirm.click();await tp.waitForTimeout(700);await tp.screenshot({path:'/tmp/precalc-q36-teacher-confirmed.png',fullPage:true});const teacherText=await tp.locator('body').innerText();
const sc=await b.newContext({storageState:'/tmp/precalc-q36-student-browser-state.json'});const sp=await sc.newPage();await sp.goto(base+'/student/lessons/1.1');await sp.waitForLoadState('networkidle');const check=sp.getByRole('button',{name:'Check lesson status',exact:true});if(await check.count()) {await check.click();await sp.waitForTimeout(1000);}await sp.screenshot({path:'/tmp/precalc-q36-student-practice.png',fullPage:true});const studentText=await sp.locator('body').innerText();fs.writeFileSync('/tmp/precalc-q36-browser-phase2.json',JSON.stringify({teacherText,studentText,buttons:await sp.getByRole('button').allTextContents(),inputs:await sp.locator('input').evaluateAll(xs=>xs.map(x=>({name:x.getAttribute('name'),type:x.getAttribute('type'),placeholder:x.getAttribute('placeholder'),aria:x.getAttribute('aria-label')}))),selects:await sp.locator('select').evaluateAll(xs=>xs.map(x=>({name:x.getAttribute('name'),aria:x.getAttribute('aria-label'),options:[...x.options].map(o=>o.text)})))},null,2));await b.close()})().catch(e=>{console.error(e);process.exit(1)})
