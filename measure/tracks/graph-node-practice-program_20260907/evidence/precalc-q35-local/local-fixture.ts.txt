export const rollbackFreshEvidence = internalMutation({
  args: { learnerId: v.id('profiles'), submissionId: v.id('activity_submissions') },
  handler: async (ctx, args) => {
    const result = await submitSelectedEvidenceHandler(ctx, args);
    if (result.delivery !== 'new' || !result.masteryApplied) throw new Error('Expected fresh mastery effects before rollback');
    const receipt = await ctx.db.query('learning_evidence_receipts').withIndex('by_learner_and_submission_id', q => q.eq('learnerId', args.learnerId).eq('submissionId', args.submissionId)).unique();
    if (!receipt) throw new Error('Expected new receipt before rollback');
    throw new Error('q35_actual_selected_effects_then_rollback');
  },
});

export const ensureFreshGuidedActivity = internalMutation({

      gradingConfig: { variantSeed: 0 },
      approval: { status: 'approved', reviewedAt: now },
      createdAt: now,
      updatedAt: now,
    });
    return { activityId };
  },
});

export const setFreshCredentialActive = internalMutation({
  args: { learnerId: v.id('profiles'), isActive: v.boolean() },
  handler: async (ctx, args) => {
    const profile = await ctx.db.get('profiles', args.learnerId);
    if (!profile) throw new Error('Missing learner');
    const credential = await ctx.db.query('auth_credentials').withIndex('by_username', q => q.eq('username', profile.username)).unique();
    if (!credential) throw new Error('Missing credential');
    await ctx.db.patch(credential._id, { isActive: args.isActive, updatedAt: Date.now() });
    return { isActive: args.isActive };
  },
});
